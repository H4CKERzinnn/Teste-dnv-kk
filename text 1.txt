local player = game:GetService("Players").LocalPlayer
local remote = game:GetService("ReplicatedStorage")
    :WaitForChild("BridgeNet")
    :WaitForChild("dataRemoteEvent")

local timeNotif = player.PlayerGui.UI.HUD.RightButtons.TimeRewards.Background.Notification
local achNotif = player.PlayerGui.UI.HUD.Left.Buttons.Container.Achievements.Background.Notification

local busyRewards = false
local busyAchievements = false

local function claimOne(tipo, index)
    for tentativa = 1, 3 do
        local args = {
            {
                {
                    "General",
                    tipo,
                    "Claim",
                    index,
                    n = 4
                },
                "\002"
            }
        }

        remote:FireServer(unpack(args))
        task.wait(0.6)

    end
end

local function claimRewards(tipo)
    for i = 1, 7 do
        claimOne(tipo, i)
        task.wait(0.5)
    end
end

local function handleRewards()
    if busyRewards then return end
    busyRewards = true

    print("Pegando rewards...")

    claimRewards("TimeRewards")
    claimRewards("DailyRewards")

    task.wait(3)
    busyRewards = false
end

local function handleAchievements()
    if busyAchievements then return end
    busyAchievements = true

    print("Pegando conquistas...")

    local args = {
        {
            {
                "General",
                "Achievements",
                "ClaimAll",
                n = 3
            },
            "\002"
        }
    }

    remote:FireServer(unpack(args))

    task.wait(2)
    busyAchievements = false
end

timeNotif:GetPropertyChangedSignal("Visible"):Connect(function()
    if timeNotif.Visible then
        handleRewards()
    end
end)

achNotif:GetPropertyChangedSignal("Visible"):Connect(function()
    if achNotif.Visible then
        handleAchievements()
    end
end)

task.spawn(function()
    task.wait(1)

    if timeNotif.Visible then
        handleRewards()
    end

    if achNotif.Visible then
        handleAchievements()
    end
end)

local function teleportIlha()
    local args = {
        {
            {
                "Player",
                "Teleport",
                "Teleport",
                "Slime Verse",
                2,
                n = 5
            },
            "\002"
        }
    }

    for i = 1, 4 do
        local success, err = pcall(function()
            game:GetService("ReplicatedStorage")
                :WaitForChild("BridgeNet")
                :WaitForChild("dataRemoteEvent")
                :FireServer(unpack(args))
        end)

        if success then
            print("✅ TP Slime Verse ok (" .. i .. "/4)")
        else
            warn("❌ TP falhou (" .. i .. "/4):", err)
        end

        task.wait(0.4)
    end
end

local player = game:GetService("Players").LocalPlayer
local remote = game:GetService("ReplicatedStorage")
    :WaitForChild("BridgeNet")
    :WaitForChild("dataRemoteEvent")

-- ⚠️ Ajuste aqui o caminho correto do reset indicator (✓ ou imagem ON)
local resetIndicator = player.PlayerGui.UI.HUD.RightButtons.TimeRewards.Background
    :FindFirstChildWhichIsA("ImageLabel", true)

-- Timer do claim (o "!" vermelho que aparece quando pode resetar)
local claimTimer = player.PlayerGui.UI.HUD.RightButtons.TimeRewards.Background:FindFirstChild("Time", true) 
                   or player.PlayerGui.UI.HUD.RightButtons.TimeRewards.Background:FindFirstChildWhichIsA("TextLabel", true)

local resetCooldown = false

local function doReset()
    if resetCooldown then return end
    resetCooldown = true

    print("🔄 Resetando TimeRewards...")

    local args = {
        {
            {
                "General",
                "TimeRewards",
                "Reset",
                n = 3
            },
            "\002"
        }
    }

    remote:FireServer(unpack(args))

    task.wait(5)   -- tempo de segurança após o reset
    resetCooldown = false
    print("✅ Reset finalizado")
end

-- Loop principal de detecção
task.spawn(function()
    while task.wait(1) do
        local shouldReset = false

        -- Verifica o indicador visual (✓ ou imagem ON)
        if resetIndicator and resetIndicator.Visible then
            shouldReset = true
        end

        -- Verifica também o timer (texto com "!" ou tempo zerado)
        if claimTimer then
            if claimTimer.Text:find("!") or claimTimer.Text == "0:00" or claimTimer.Text == "00:00" then
                shouldReset = true
            end
        end

        if shouldReset then
            doReset()
        end
    end
end)

-- Check inicial
task.spawn(function()
    task.wait(1.5)
    
    local shouldReset = false
    
    if resetIndicator and resetIndicator.Visible then
        shouldReset = true
    end
    
    if claimTimer then
        if claimTimer.Text:find("!") or claimTimer.Text == "0:00" or claimTimer.Text == "00:00" then
            shouldReset = true
        end
    end
    
    if shouldReset then
        doReset()
    end
end)

print("✅ Script de Reset Separado carregado!")
