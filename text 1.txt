local remote = game:GetService("ReplicatedStorage")
	:WaitForChild("ffrostflame_bridgenet2@1.0.0")
	:WaitForChild("dataRemoteEvent")

local potions = {"Ghost","Drop","Luck","Damage","Energy"}

while true do
	-- 🔥 1. Usar potions
	for _, name in ipairs(potions) do
		local args = {
			{
				{
					"ItemSystem",
					"Use",
					name.."Potion4",
					n = 3
				},
				"\002"
			}
		}
		remote:FireServer(unpack(args))
		task.wait(1)
	end

	task.wait(30)

	-- 🐉 2. DragonBox
	do
		local args = {
			{
				{
					"ItemSystem",
					"UseAll",
					"DragonBox",
					n = 3
				},
				"\002"
			}
		}
		remote:FireServer(unpack(args))
	end

	task.wait(45)

	-- ⚗️ 3. Converter potion
	for _, name in ipairs(potions) do
		local args = {
			{
				{
					"ExchangeSystem",
					"Make",
					"Potion",
					"Tier 3",
					name.."Potion3",
					1,
					n = 6
				},
				"\002"
			}
		}
		remote:FireServer(unpack(args))
		task.wait(1)
	end

	task.wait(60)

	-- 🔁 aqui automaticamente volta pro início
end
