local remote = game:GetService("ReplicatedStorage")
	:WaitForChild("ffrostflame_bridgenet2@1.0.0")
	:WaitForChild("dataRemoteEvent")

-- 🔥 Loop 1: Usar potions
task.spawn(function()
	local potions = {"Ghost","Drop","Luck","Damage","Energy"}

	while true do
		for _, name in ipairs(potions) do
			local args = {
				{
					{
						"ItemSystem",
						"Use",
						name.."Potion4",
						n = 3
					},
					"\002"
				}
			}
			remote:FireServer(unpack(args))
			task.wait(1)
		end
		task.wait(30)
	end
end)

-- 🔥 Loop 2: DragonBox
task.spawn(function()
	while true do
		local args = {
			{
				{
					"ItemSystem",
					"UseAll",
					"DragonBox",
					n = 3
				},
				"\002"
			}
		}
		remote:FireServer(unpack(args))
		task.wait(45)
	end
end)

-- 🔥 Loop 3: Converter potion
task.spawn(function()
	local potions = {"Damage","Ghost","Luck","Drop","Energy"}

	while true do
		for _, name in ipairs(potions) do
			local args = {
				{
					{
						"ExchangeSystem",
						"Make",
						"Potion",
						"Tier 3",
						name.."Potion3",
						1,
						n = 6
					},
					"\002"
				}
			}
			remote:FireServer(unpack(args))
			task.wait(1)
		end
		task.wait(60)
	end
end)
