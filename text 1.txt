local remote = game:GetService("ReplicatedStorage")
	:WaitForChild("ffrostflame_bridgenet2@1.0.0")
	:WaitForChild("dataRemoteEvent")

local potions = {
	"Ghost",
	"Drop",
	"Luck",
	"Damage",
	"Energy"
}

local function usePotion(name)
	local args = {
		{
			{
				"ItemSystem",
				"Use",
				name.."Potion4",
				n = 3
			},
			"\002"
		}
	}
	remote:FireServer(unpack(args))
end

while true do
	for _, name in ipairs(potions) do
		usePotion(name)
		task.wait(1) -- 🔥 delay pra não bugar
	end

	task.wait(60) -- ⏱️ 1 minuto pra repetir tudo
end

-- Potion tier 3 -> 4

local remote = game:GetService("ReplicatedStorage")
	:WaitForChild("ffrostflame_bridgenet2@1.0.0")
	:WaitForChild("dataRemoteEvent")

local potions = {
	"Damage",
	"Ghost",
	"Luck",
	"Drop",
	"Energy"
}

while true do
	for _, name in ipairs(potions) do
		local args = {
			{
				{
					"ExchangeSystem",
					"Make",
					"Potion",
					"Tier 3",
					name .. "Potion3",
					1,
					n = 6
				},
				"\002"
			}
		}

		remote:FireServer(unpack(args))
		task.wait(1) -- 🔥 espera 1 segundo entre cada envio
	end

	task.wait(60) -- espera 1 minuto pra repetir tudo
end
