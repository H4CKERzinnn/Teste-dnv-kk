local player = game:GetService("Players").LocalPlayer
local remote = game:GetService("ReplicatedStorage")
    :WaitForChild("BridgeNet")
    :WaitForChild("dataRemoteEvent")

local timeNotif = player.PlayerGui.UI.HUD.RightButtons.TimeRewards.Background.Notification
local achNotif = player.PlayerGui.UI.HUD.Left.Buttons.Container.Achievements.Background.Notification

local busyRewards = false
local busyAchievements = false

local function claimOne(tipo, index)
    for tentativa = 1, 3 do
        local args = {
            {
                {
                    "General",
                    tipo,
                    "Claim",
                    index,
                    n = 4
                },
                "\002"
            }
        }

        remote:FireServer(unpack(args))
        task.wait(0.6)

    end
end

local function claimRewards(tipo)
    for i = 1, 7 do
        claimOne(tipo, i)
        task.wait(0.5)
    end
end

local function handleRewards()
    if busyRewards then return end
    busyRewards = true

    print("Pegando rewards...")

    claimRewards("TimeRewards")
    claimRewards("DailyRewards")

    task.wait(3)
    busyRewards = false
end

local function handleAchievements()
    if busyAchievements then return end
    busyAchievements = true

    print("Pegando conquistas...")

    local args = {
        {
            {
                "General",
                "Achievements",
                "ClaimAll",
                n = 3
            },
            "\002"
        }
    }

    remote:FireServer(unpack(args))

    task.wait(2)
    busyAchievements = false
end

timeNotif:GetPropertyChangedSignal("Visible"):Connect(function()
    if timeNotif.Visible then
        handleRewards()
    end
end)

achNotif:GetPropertyChangedSignal("Visible"):Connect(function()
    if achNotif.Visible then
        handleAchievements()
    end
end)

task.spawn(function()
    task.wait(1)

    if timeNotif.Visible then
        handleRewards()
    end

    if achNotif.Visible then
        handleAchievements()
    end
end)

local Players    = game:GetService("Players")
local RunService = game:GetService("RunService")

local player    = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local rootPart  = character:WaitForChild("HumanoidRootPart")

local ENEMIES_FOLDER = workspace:WaitForChild("Client"):WaitForChild("Enemies")

-- 📍 Coordenadas
local TRIAL_POS = Vector3.new(-4250, 1035, -11407)
local ILHA_POS  = Vector3.new(624, 807, -827)

-- Config
local cfg = {
    autoTPEnabled = false, -- começa desligado
    tpInterval    = 0.6,
    tpRange       = 600,
    tpOffset      = 4,
    detectRange   = 50 -- distância pra detectar chegada na trial
}

local currentIndex = 1
local lastTP = 0
local jaTeleportou = false

-- 🔍 Checar se está perto de uma posição
local function isNear(pos, range)
    return (rootPart.Position - pos).Magnitude <= range
end

-- 👹 Pegar inimigos
local function getLocalEnemies()
    local list = {}
    local origin = rootPart.Position

    for _, m in ipairs(ENEMIES_FOLDER:GetChildren()) do
        if not m:IsA("Model") then continue end

        local tp = m:FindFirstChild("HumanoidRootPart")
            or m:FindFirstChild("Torso")
            or m.PrimaryPart
            or m:FindFirstChildWhichIsA("BasePart")

        if not tp then continue end
        if tp.Position.Y < -200 then continue end
        if (tp.Position - origin).Magnitude > cfg.tpRange then continue end

        table.insert(list, tp)
    end

    return list
end

-- 🤖 AUTO FARM
RunService.Heartbeat:Connect(function()
    -- 🟢 ATIVA quando chega na trial
    if isNear(TRIAL_POS, cfg.detectRange) then
        cfg.autoTPEnabled = true
    end

    if not cfg.autoTPEnabled then return end
    if tick() - lastTP < cfg.tpInterval then return end

    lastTP = tick()

    local enemies = getLocalEnemies()
    if #enemies == 0 then return end

    if currentIndex > #enemies then
        currentIndex = 1
    end

    local target = enemies[currentIndex]

    if target then
        local dir = (rootPart.Position - target.Position).Unit
        rootPart.CFrame = CFrame.new(target.Position + dir * cfg.tpOffset, target.Position)
    end

    currentIndex = (currentIndex % #enemies) + 1
end)

-- 📊 PEGAR WAVE
local waveLabel = player.PlayerGui
    :WaitForChild("UI")
    :WaitForChild("HUD")
    :WaitForChild("Gamemodes")
    :WaitForChild("Trial Easy")
    :WaitForChild("Main")
    :WaitForChild("Wave")
    :WaitForChild("Value")

local function getCurrentWave()
    local text = waveLabel.Text or ""
    local num = text:match("(%d+)%s*/") or text:match("(%d+)/")
    return tonumber(num) or 0
end

-- 🌍 TP Slime
local function teleportSlime()
    local args = {
        {
            {
                "Player",
                "Teleport",
                "Teleport",
                "Slime Verse",
                1,
                n = 5
            },
            "\002"
        }
    }

    for i = 1, 5 do
        pcall(function()
            game:GetService("ReplicatedStorage")
                :WaitForChild("BridgeNet")
                :WaitForChild("dataRemoteEvent")
                :FireServer(unpack(args))
        end)
        task.wait(0.3)
    end
end

-- 🎯 DETECTOR DE WAVE
waveLabel:GetPropertyChangedSignal("Text"):Connect(function()
    local current = getCurrentWave()

    -- reset
    if current <= 5 then
        jaTeleportou = false
    end

    -- 🚪 SAÍDA NA WAVE 40
    if not jaTeleportou and current >= 40 then
        jaTeleportou = true

        print("🚪 Saindo da trial...")

        -- 🔴 DESATIVA FARM
        cfg.autoTPEnabled = false

        teleportSlime()

        task.wait(2)

        -- 📍 TP PRA ILHA
        character = player.Character or player.CharacterAdded:Wait()
        rootPart = character:WaitForChild("HumanoidRootPart")

        rootPart.CFrame = CFrame.new(ILHA_POS)
    end
end)

-- 🔄 Respawn
player.CharacterAdded:Connect(function(c)
    character = c
    rootPart = c:WaitForChild("HumanoidRootPart")
    lastTP = tick()
    currentIndex = 1
end)

print("🔥 Sistema FULL AUTO carregado!")
