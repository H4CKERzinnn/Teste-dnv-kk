local player = game:GetService("Players").LocalPlayer
local remote = game:GetService("ReplicatedStorage")
    :WaitForChild("BridgeNet")
    :WaitForChild("dataRemoteEvent")

local timeNotif = player.PlayerGui.UI.HUD.RightButtons.TimeRewards.Background.Notification
local achNotif = player.PlayerGui.UI.HUD.Left.Buttons.Container.Achievements.Background.Notification

local busyRewards = false
local busyAchievements = false

local function claimOne(tipo, index)
    for tentativa = 1, 3 do
        local args = {
            {
                {
                    "General",
                    tipo,
                    "Claim",
                    index,
                    n = 4
                },
                "\002"
            }
        }

        remote:FireServer(unpack(args))
        task.wait(0.6)

    end
end

local function claimRewards(tipo)
    for i = 1, 7 do
        claimOne(tipo, i)
        task.wait(0.5)
    end
end

local function handleRewards()
    if busyRewards then return end
    busyRewards = true

    print("Pegando rewards...")

    claimRewards("TimeRewards")
    claimRewards("DailyRewards")

    task.wait(3)
    busyRewards = false
end

local function handleAchievements()
    if busyAchievements then return end
    busyAchievements = true

    print("Pegando conquistas...")

    local args = {
        {
            {
                "General",
                "Achievements",
                "ClaimAll",
                n = 3
            },
            "\002"
        }
    }

    remote:FireServer(unpack(args))

    task.wait(2)
    busyAchievements = false
end

timeNotif:GetPropertyChangedSignal("Visible"):Connect(function()
    if timeNotif.Visible then
        handleRewards()
    end
end)

achNotif:GetPropertyChangedSignal("Visible"):Connect(function()
    if achNotif.Visible then
        handleAchievements()
    end
end)

task.spawn(function()
    task.wait(1)

    if timeNotif.Visible then
        handleRewards()
    end

    if achNotif.Visible then
        handleAchievements()
    end
end)

local remote = game:GetService("ReplicatedStorage")
	:WaitForChild("BridgeNet")
	:WaitForChild("dataRemoteEvent")

local rolls = {
	"Swords Banner",
	"Haki",
	"Fruit",
	"Dragon Power",
	"Race",
	"Slime Power",
	"Primordial Demon"
}

while true do
	for _, rollName in ipairs(rolls) do
		local args = {
			{
				{
					"General",
					rollName == "Swords Banner" and "Banner" or "Gacha",
					"Roll",
					rollName,
					rollName == "Swords Banner" and nil or {},
					n = rollName == "Swords Banner" and 4 or 5
				},
				"\002"
			}
		}

		remote:FireServer(unpack(args))
		task.wait(0.004) -- tempo entre cada roll
	end
end


-- ⏰ CONTROLE POR HORÁRIO

local jaAtivouMinuto = ""
local jaTeleportouMinuto = ""

local function teleportIlha()
    local args = {
        {
            {
                "Player",
                "Teleport",
                "Teleport",
                "Slime Verse",
                1,
                n = 5
            },
            "\002"
        }
    }

    for i = 1, 5 do
        pcall(function()
            game:GetService("ReplicatedStorage")
                :WaitForChild("BridgeNet")
                :WaitForChild("dataRemoteEvent")
                :FireServer(unpack(args))
        end)
        task.wait(0.3)
    end
end

local function teleportCoordenada()
    local char = player.Character or player.CharacterAdded:Wait()
    local hrp = char:WaitForChild("HumanoidRootPart")

    hrp.CFrame = CFrame.new(624, 807, -827)
end

task.spawn(function()
    while true do
        local minuto = os.date("*t").min

        -- 🟢 ATIVA FARM (15 e 45)
        if (minuto == 15 or minuto == 45) and jaAtivouMinuto ~= tostring(minuto) then
            jaAtivouMinuto = tostring(minuto)

            print("🟢 Ativando auto farm (", minuto, ")")

            cfg.autoTPEnabled = true
        end

        -- 🔴 DESATIVA + TP (19 e 49)
        if (minuto == 19 or minuto == 49) and jaTeleportouMinuto ~= tostring(minuto) then
            jaTeleportouMinuto = tostring(minuto)

            print("🔴 Desligando farm + TP (", minuto, ")")

            cfg.autoTPEnabled = false

            teleportIlha()

            task.wait(2)

            teleportCoordenada()
        end

        task.wait(5) -- checa a cada 5 segundos
    end
end)
