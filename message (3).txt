-- Wave Label
local waveLabel = nil
task.spawn(function()
    local ok, result = pcall(function()
        return player.PlayerGui
            :WaitForChild("UI",10):WaitForChild("HUD",10)
            :WaitForChild("Gamemodes",10):WaitForChild("Trial Easy",10)
            :WaitForChild("Main",10):WaitForChild("Wave",10):WaitForChild("Value",10)
    end)
    if ok then waveLabel = result end
end)

local function getCurrentWave()
    if not waveLabel then return 0 end
    local t = waveLabel.Text or ""
    return tonumber(t:match("(%d+)%s*/") or t:match("(%d+)/")) or 0
end

local function tentarTeleport()
    local args = {{{ "Player","Teleport","Teleport","Dragon Verse",1,n=5},"\002"}}
    for i=1,5 do
        pcall(function()
            game:GetService("ReplicatedStorage")
                :WaitForChild("BridgeNet",2):WaitForChild("dataRemoteEvent",2)
                :FireServer(unpack(args))
        end)
        task.wait(0.35)
    end
end

local function getLocalEnemies()
    local list,origin = {},rootPart.Position
    for _,m in ipairs(ENEMIES_FOLDER:GetChildren()) do
        if not m:IsA("Model") then continue end
        local tp = m:FindFirstChild("HumanoidRootPart") or m:FindFirstChild("Torso")
               or m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
        if not tp or tp.Position.Y < -200 then continue end
        if (tp.Position-origin).Magnitude > cfg.tpRange then continue end
        table.insert(list,tp)
    end
    return list
end

RunService.Heartbeat:Connect(function()
    if not cfg.autoTPEnabled then return end
    if tick()-lastTP < cfg.tpInterval then return end
    lastTP = tick()
    local e = getLocalEnemies()
    if #e==0 then return end
    if currentIndex>#e then currentIndex=1 end
    local tp=e[currentIndex]
    if tp then
        local dir=(rootPart.Position-tp.Position).Unit
        rootPart.CFrame=CFrame.new(tp.Position+dir*cfg.tpOffset,tp.Position)
    end
    currentIndex=(currentIndex%#e)+1
end)

player.CharacterAdded:Connect(function(c)
    character=c; rootPart=c:WaitForChild("HumanoidRootPart")
    lastTP=tick(); currentIndex=1
end)

local function slider(par,mn,mx,cur,step,cb)
    local tr=Instance.new("Frame"); tr.Size=UDim2.new(1,-16,0,5)
    tr.Position=UDim2.new(0,8,1,-14); tr.BackgroundColor3=Color3.fromRGB(38,38,58)
    tr.BorderSizePixel=0; tr.ZIndex=par.ZIndex+1; tr.Parent=par; cr(tr,3)
    local fl=Instance.new("Frame"); fl.Size=UDim2.new((cur-mn)/(mx-mn),0,1,0)
    fl.BackgroundColor3=C.accent; fl.BorderSizePixel=0; fl.ZIndex=tr.ZIndex+1; fl.Parent=tr; cr(fl,3)
    local hd=Instance.new("TextButton"); hd.Size=UDim2.new(0,18,0,18)
    hd.AnchorPoint=Vector2.new(0.5,0.5)
    hd.Position=UDim2.new((cur-mn)/(mx-mn),0,0.5,0)
    hd.BackgroundColor3=Color3.fromRGB(255,255,255); hd.Text=""
    hd.ZIndex=tr.ZIndex+2; hd.Parent=tr; cr(hd,9)
    local v=cur; local dr=false
    local function upd(x)
        local rel=math.clamp((x-tr.AbsolutePosition.X)/tr.AbsoluteSize.X,0,1)
        v=math.clamp(math.floor((mn+(mx-mn)*rel)/step+0.5)*step,mn,mx)
        local p=(v-mn)/(mx-mn)
        fl.Size=UDim2.new(p,0,1,0); hd.Position=UDim2.new(p,0,0.5,0); cb(v)
    end
    hd.MouseButton1Down:Connect(function() dr=true end)
    tr.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.Touch then dr=true; upd(i.Position.X) end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1
        or i.UserInputType==Enum.UserInputType.Touch then dr=false end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if not dr then return end
        if i.UserInputType==Enum.UserInputType.MouseMovement
        or i.UserInputType==Enum.UserInputType.Touch then upd(i.Position.X) end
    end)
end
